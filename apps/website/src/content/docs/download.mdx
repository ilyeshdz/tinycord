<script>
;(async function() {
  var container = document.getElementById('downloads-container');
  try {
    var res = await fetch('https://api.github.com/repos/ilyeshdz/tinycord/releases/latest');
    if (!res.ok) throw new Error(String(res.status));
    var release = await res.json();
    var version = release.tag_name.replace(/^v/, '');
    var assets = release.assets;

    function match(re) {
      return assets.find(function(a) { return re.test(a.name); }) || null;
    }

    var dmgArm   = match(/macOS.*arm64.*\.dmg$/i);
    var dmgX86   = match(/macOS.*x86_64.*\.dmg$/i);
    var exeX86   = match(/Windows.*x86_64.*\.exe$/i);
    var exeArm   = match(/Windows.*arm64.*\.exe$/i);

    var plat = navigator.platform.toLowerCase();
    var isMac = plat.includes('mac');
    var isWin = plat.includes('win');

    function card(label, arch, url, primary) {
      var cls = primary ? 'download-card primary' : 'download-card';
      var badge = primary ? '<span class="platform-badge">Recommended</span>' : '';
      return '<a href="' + url + '" class="' + cls + '">' + badge + '<span class="platform">' + label + '</span><span class="meta">' + arch + '</span><span class="action">Download \u2193</span></a>';
    }

    var html = '<p style="margin-bottom:0.5rem"><strong>Version ' + version + '</strong></p>';
    html += '<div class="download-grid">';

    // macOS
    if (dmgArm) html += card('macOS', 'Apple Silicon', dmgArm.browser_download_url, isMac);
    if (dmgX86) html += card('macOS', 'Intel', dmgX86.browser_download_url, isMac && !dmgArm);
    if (!dmgArm && !dmgX86) {
      // fallback if no DMG yet - show raw archive
      var rawMac = match(/macOS.*\.tar\.gz$/i);
      if (rawMac) html += card('macOS', 'Raw binary (no DMG yet)', rawMac.browser_download_url, isMac);
    }

    // Windows
    if (exeX86) html += card('Windows', 'x86_64', exeX86.browser_download_url, isWin);
    if (exeArm) html += card('Windows', 'ARM64', exeArm.browser_download_url, isWin && !exeX86);
    if (!exeX86 && !exeArm) {
      var rawWin = match(/Windows.*\.zip$/i);
      if (rawWin) html += card('Windows', 'Raw binary (no installer yet)', rawWin.browser_download_url, isWin);
    }

    html += '</div>';
    html += '<p><a href="' + release.html_url + '">View all release assets on GitHub \u2192</a></p>';
    container.innerHTML = html;
  } catch(e) {
    container.innerHTML = '<p>Could not load release info. <a href="https://github.com/ilyeshdz/tinycord/releases/latest">Download from GitHub Releases \u2192</a></p>';
  }
})();
</script>
